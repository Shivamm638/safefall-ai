# SafeFall AI - runtime dependencies for the deployed Streamlit dashboard.
#
# TensorFlow is deliberately NOT listed here. The dashboard runs the trained
# network through the pure-NumPy engine in src/numpy_inference.py, which loads
# the exported Keras weights and reproduces them to within 1e-6 (see
# results/engine_parity_check.json). That keeps the deployed app well inside
# the 1 GB memory limit of Streamlit Community Cloud.
#
# To reproduce training instead, install requirements-dev.txt.

# streamlit-webrtc 0.77 requires streamlit >= 1.51, so the old 1.49.1 pin made
# the two unresolvable. 1.62.0 is the version this project is actually
# developed and tested against locally.
streamlit==1.62.0
pandas==2.3.3

# Real-time camera on the DEPLOYED app. OpenCV can only open a camera attached
# to the machine running the process, which in the cloud is a server with none -
# so the deployed app could otherwise offer still snapshots only. streamlit-webrtc
# streams the *viewer's* browser camera to the app instead, giving genuine
# real-time monitoring from wherever the visitor is. Pure-Python wheels, and `av`
# ships cp314 wheels, so this installs on the interpreter Streamlit Cloud provides.
streamlit-webrtc==0.77.0
plotly==6.9.0
pillow==11.3.0

# ---------------------------------------------------------------------------
# MediaPipe
# ---------------------------------------------------------------------------
# MediaPipe split into two incompatible eras: <= 0.10.21 exposes
# `mp.solutions.pose` but publishes no wheels above Python 3.12, while every
# release that supports 3.13/3.14 removed `mp.solutions` entirely.
#
# src/pose_backend.py detects which API is present and adapts, so either era
# runs. For INSTALLATION there is no such ambiguity: the 0.10.33 wheel is
# tagged `py3-none-manylinux_2_28_x86_64`, i.e. it carries no interpreter
# version constraint at all, so one pin covers every Python this project runs
# on - 3.12 locally and 3.14 on Streamlit Cloud. Pinning the legacy release
# behind a marker bought nothing and dragged protobuf < 5 with it, which modern
# Streamlit (protobuf >= 5.26.1) cannot satisfy.
mediapipe==0.10.33

# protobuf must be pinned. mediapipe 0.10.33 declares no constraint, but its
# generated code still calls MessageFactory.GetPrototype, which protobuf 6
# removed. Left unpinned, pip takes streamlit's `protobuf<8` and installs 7.x,
# and every pose call then dies with:
#     AttributeError: 'MessageFactory' object has no attribute 'GetPrototype'
#
# Verified directly: 5.29.5 detects all 33 landmarks, 6.33.1 raises. 5.29.5
# also satisfies streamlit's protobuf >= 5.26.1.
protobuf==5.29.5

# numpy is the one dependency that genuinely differs by interpreter: 1.26.4
# publishes no cp314 wheels, so Streamlit Cloud must take the 2.x line.
numpy==1.26.4; python_version < "3.13"
numpy>=2.1; python_version >= "3.13"

# MediaPipe's drawing_utils imports matplotlib at module load, so the pose
# overlay silently depends on it. mediapipe declares it, but pinning it here
# means a future release cannot drop it out from under us. 3.11.1 ships cp312
# and cp314 wheels, so no marker is needed.
matplotlib==3.11.1

# OpenCV comes in via mediapipe, which requires `opencv-contrib-python`
# specifically - adding `opencv-python-headless` alongside it installs a second,
# conflicting copy of the same `cv2` module. The non-headless build needs
# libgl1, which is why packages.txt exists.
#
# The exact version is left to the resolver, but capped below 5.0. Unpinned,
# pip takes the newest release, which is now the freshly published OpenCV 5.0 -
# a major version this project has never been run against. The cap keeps the
# deployed app on the same OpenCV 4 API line used in development (4.11 locally,
# 4.14 on Python 3.14, which is the newest 4.x publishing cp314 wheels).
opencv-contrib-python<5
