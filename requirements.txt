# SafeFall AI - runtime dependencies for the deployed Streamlit dashboard.
#
# TensorFlow is deliberately NOT listed here. The dashboard runs the trained
# network through the pure-NumPy engine in src/numpy_inference.py, which loads
# the exported Keras weights and reproduces them to within 1e-6 (see
# results/engine_parity_check.json). That keeps the deployed app well inside
# the 1 GB memory limit of Streamlit Community Cloud.
#
# To reproduce training instead, install requirements-dev.txt.

streamlit==1.49.1
pandas==2.3.3
plotly==6.9.0
pillow==11.3.0

# ---------------------------------------------------------------------------
# MediaPipe: two incompatible eras, selected by interpreter
# ---------------------------------------------------------------------------
# mediapipe <= 0.10.21 exposes `mp.solutions.pose` but publishes no wheels
# above Python 3.12. Every release that DOES support 3.13/3.14 removed
# `mp.solutions` entirely. No single version spans both, so the version is
# chosen by environment marker and src/pose_backend.py adapts to whichever
# API it finds - the legacy solutions API, or mediapipe.tasks.
#
# Both drive the same BlazePose network. Verified identical on the bundled
# samples: same predicted class every time, confidences agreeing to 3-4
# decimals, trunk angles within 0.1 degrees.
mediapipe==0.10.21; python_version < "3.13"
mediapipe==0.10.33; python_version >= "3.13"

# numpy: mediapipe 0.10.21 requires <2; 0.10.33 has no upper bound.
numpy==1.26.4; python_version < "3.13"
numpy>=2.1; python_version >= "3.13"

# protobuf: pinned only for the legacy build, which requires <5. The newer
# mediapipe no longer constrains it, so pip is left to resolve it there.
protobuf==4.25.9; python_version < "3.13"

# MediaPipe's drawing_utils imports matplotlib at module load, so the pose
# overlay silently depends on it. mediapipe declares it, but pinning it here
# means a future release cannot drop it out from under us.
matplotlib==3.11.1; python_version < "3.13"
matplotlib; python_version >= "3.13"

# OpenCV is intentionally NOT pinned: mediapipe requires `opencv-contrib-python`
# specifically, and adding `opencv-python-headless` alongside installs a second,
# conflicting copy of the same `cv2` module. The non-headless build needs
# libgl1, which is why packages.txt exists.
